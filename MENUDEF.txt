// ============================================================================
// RS_QuestHands -- options
//
// STRUCTURE, because this grows. Everything splits three ways:
//
//   Hands        -- where each of the four pieces sits. Set once, true for
//                   every weapon in every set, because it is a property of the
//                   hand meshes and the controllers.
//   Steadying    -- the engine's own two-hand behaviour and our guard on it.
//   Weapon sets  -- ONE SUBMENU PER SET. Reload feel differs between sets
//                   because their guns differ, so the gesture distances and the
//                   insert count live here rather than globally.
//
// Owner, 2026-10-03: "expect an options menu for all sets we do here." Adding a
// set means adding a Submenu line below and one OptionMenu block, not rewriting
// this file.
//
// Slider units for hand placement are MODEL units in the hook's frame: X right,
// Y up, Z back toward the player, 1 unit = 1 cm times the MODELDEF scale.
// Gesture distances are REAL METRES between the hands, metric-corrected -- see
// zscript/qh_metric.zs for why those are not the same thing.
// ============================================================================

AddOptionMenu "OptionsMenu"
{
	Submenu "Quest Hands", "QH_Options"
}
