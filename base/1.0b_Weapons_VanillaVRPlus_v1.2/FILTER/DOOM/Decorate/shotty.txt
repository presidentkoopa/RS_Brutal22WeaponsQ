ACTOR VR_Shotgun : Shotgun
{
  Weapon.AmmoUse 0
  Weapon.AmmoGive1 8
  Weapon.AmmoGive2 0
  Weapon.AmmoType1 "VR_Shell"
  Weapon.AmmoType2 "VR_ShotLoaded"
  
  States
  {
  Ready:
    SHTG A 1 A_WeaponReady(WRF_ALLOWRELOAD)
    Loop

  Deselect:
	TNT1 A 0 A_JumpIf(GetCVAR("evw_fastswitch")==0, 2)
    SHTG A 1 A_Lower(60)
    Loop

  Select:
	TNT1 A 0 A_JumpIf(GetCVAR("evw_fastswitch")==0, 2)
    SHTG A 1 A_Raise(60)
    Loop

  Fire:
    TNT1 A 0 A_JumpIfInventory("VR_ShotLoaded", 1, "CommitExtremeViolence")
    Goto Reload

  CommitExtremeViolence:
    TNT1 A 0 A_JumpIfInventory("VR_ShotLoaded", 1, "ACommitExtremeViolence")
    TNT1 A 0 A_PlaySound("AKEMPT", CHAN_AUTO)
    Goto Ready

  ACommitExtremeViolence:
    TNT1 A 0 A_TakeInventory("VR_ShotLoaded", 1)
    SHTG A 0 A_FireBullets(5.6, 0, 7, 5, "vrshotpuff")
    SHTG A 0 A_PlaySound("wpn/shotgun1", CHAN_WEAPON)
    SHTG A 0 A_GunFlash
    SHTG BCDEFG 2
    SHTG HI 1
	SHTG A 0 A_PlaySound("shotpump", CHAN_WEAPON)
	SHTG JKLMNOPQRSTUVWXYZ 1
    Goto Ready

  Flash:
    TNT1 AAAAAA 0 A_FireProjectile("BulletTracer",random(-3,3),0,0,0,FPF_NOAUTOAIM,random(-1,1))
    SHTF ABC 1 Bright A_Light2
    SHTF DEF 1 Bright A_Light1
    Goto LightDone

  Spawn:
    SHOT A -1
    Stop

  // ========================
  // RELOAD (NON-INTERRUPT)
  // ========================
  Reload:
    // If full, don't reload
    TNT1 A 0 A_JumpIfInventory("VR_ShotLoaded", 8, "Ready")
    // If no spare shells, don't reload
    TNT1 A 0 A_JumpIfInventory("VR_Shell", 1, "StartReload")
    Goto OutOfAmmo

  StartReload:
    // Start animation
    SHTG A 3
    Goto ReloadLoop

  ReloadLoop:
    // INTERRUPT RELOAD IF PLAYER FIRES
    SHTG A 1 A_WeaponReady(WRF_ALLOWRELOAD)
    // Check if full
    TNT1 A 0 A_JumpIfInventory("VR_ShotLoaded", 8, "ReloadEnd")
    // Check if out of ammo
    TNT1 A 0 A_JumpIfInventory("VR_Shell", 1, "InsertShell")
    Goto OutOfAmmo

  InsertShell:
    // Insert shell animation + sound
    SHTG RQPONMLKJ 1
    TNT1 A 0 A_PlaySound("Shotgun/Reload", CHAN_AUTO)
    TNT1 A 0 A_GiveInventory("VR_ShotLoaded", 1)
    TNT1 A 0 A_TakeInventory("VR_Shell", 1)
    // Loop back until full or empty
    Goto ReloadLoop

  ReloadEnd:
    // Finish reload animation
    SHTG A 3
    Goto Ready

  OutOfAmmo:
    TNT1 A 0 A_PlaySound("AKEMPT", CHAN_AUTO)
    Goto Ready
  }
}

ACTOR VR_ShotLoaded : Ammo
{  
  Inventory.Amount 0
  Inventory.MaxAmount 8
  Ammo.BackpackAmount 0
  Ammo.BackpackMaxAmount 8
  Inventory.Icon "SHOTA0"
}

ACTOR VR_SuperShotgun : SuperShotgun
{
  Weapon.AmmoType "VR_Shell"
  States
  {
  Ready:
    SHT2 A 1 A_WeaponReady(WRF_ALLOWRELOAD)
    Loop
  Deselect:
	TNT1 A 0 A_JumpIf(GetCVAR("evw_fastswitch")==0, 2)
    SHT2 AA 1 A_Lower(60)
    Loop
  Select:
TNT1 A 0 A_JumpIf(GetCVAR("evw_fastswitch")==0, 2)
    SHT2 AA 1 A_Raise(60)
    Loop
  Fire:
	TNT1 A 0 A_JumpIfInventory ("VR_SSGLoaded",1,"CommitExtremeViolence")
	TNT1 A 0 A_JumpIfInventory ("VR_SSGAlt",1,"ACommitExtremeViolence")
	Goto Reload
  CommitExtremeViolence:
	TNT1 A 0 A_TakeInventory("VR_SSGLoaded", 1)
	TNT1 A 0 A_TakeInventory("VR_SSGAlt", 2)
    SHT2 A 0 A_FireBullets (11.2, 7.1, 20, 5, "vrssgpuff")
    SHT2 A 0 A_PlaySound ("wpn/1shotgun", CHAN_AUTO)
    SHT2 A 2 A_GunFlash
    SHT2 BCDEFGH 2
	Goto Reload
	
  AltFire:
	TNT1 A 0 A_JumpIfInventory ("VR_SSGAlt",1,"ACommitExtremeViolence")
	Goto Reload
  ACommitExtremeViolence:
	TNT1 A 0 A_TakeInventory("VR_SSGLoaded", 1)
	TNT1 A 0 A_TakeInventory("VR_SSGAlt", 1)
	TNT1 A 0 A_TakeInventory("VR_Shell", 1)
    SHTG A 0 A_FireBullets (5.6, 0, 7, 5, "vrshotpuff")
    SHTG A 0 A_PlaySound ("wpn/shotgun1", CHAN_WEAPON)
    SHT2 A 2 A_GunFlash
    SHT2 BC 2
	SHT2 DEFGH 2 A_WeaponReady(WRF_ALLOWRELOAD)
	SHT2 I 10 A_WeaponReady(WRF_ALLOWRELOAD) // cooldown frame
	
	// Auto-reload trigger if both chambers are empty
    TNT1 A 0 A_JumpIfInventory("VR_SSGLoaded", 1, "Ready") // if still loaded, stay ready
    TNT1 A 0 A_JumpIfInventory("VR_SSGAlt", 1, "Ready")    // if alt chamber still loaded, stay ready
    Goto Reload
	
  Reload:
	TNT1 A 0 A_JumpIfInventory ("VR_SSGLoaded",1,"Ready")
    SHT2 I 0 A_CheckReload
    SHT2 I 2 A_OpenShotgun2
    SHT2 JKLMNOPQR 3 
	TNT1 A 0 A_GiveInventory("VR_SSGLoaded", 1)
	TNT1 A 0 A_GiveInventory("VR_SSGAlt", 2)
    SHT2 S 2 A_LoadShotgun2
    SHT2 TUV 3
    SHT2 X 2 A_CloseShotgun2 
    SHT2 Y 2
    SHT2 Y 1 A_ReFire
    Goto Ready
  Flash:
    TNT1 AAAAAA 0 A_FireProjectile("BulletTracer",random(-3,3),0,0,0,FPF_NOAUTOAIM,random(-1,1))
    SHT4 AB 1 Bright A_Light2
    SHT4 CD 1 Bright A_Light1
    Goto LightDone
  Spawn:
    SGN2 A -1
    Stop
  }
}

{
inventory.maxamount 1
}
actor VR_SSGAlt : Inventory
{
inventory.maxamount 2
}
