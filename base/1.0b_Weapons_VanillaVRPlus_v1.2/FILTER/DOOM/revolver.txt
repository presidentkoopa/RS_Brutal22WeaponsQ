actor VR_Revolver: Weapon
{
		Weapon.SelectionOrder 1898
		Weapon.AmmoUse 0
		Weapon.AmmoGive1 48
		Weapon.AmmoGive2 0
		Weapon.AmmoType1 "Clip"
		Weapon.AmmoType2 "VR_RevLoaded"
		Tag "Revolver"
	States
	{
	Ready:
		REVL A 1 A_WeaponReady(WRF_ALLOWRELOAD)
		Loop
	Deselect:
		REVL A 1 A_Lower
		Loop
	Select:
		REVL A 1 A_raise
		REVL AAAAAAAAAAAAAAAAAAAAAAAA 1 A_Raise //GUN IS SCREAMING !!
	Fire:
		TNT1 A 0 A_JumpIfInventory ("VR_RevLoaded",1,"CommitExtremeViolence")
		Goto Reload
  CommitExtremeViolence:
		REVL BCD 1
        TNT1 A 0 A_Gunflash
		TNT1 A 0 A_FireBullets (0, 0, -1, 30, "postalpuff")
		TNT1 A 0 A_playsound("wpn/revolver", CHAN_WEAPON)
		TNT1 A 0 A_TakeInventory("VR_RevLoaded",1)
		//gun has been fired
		REVL E 2
		REVL F 2
		REVL GHIJKLMNOP 1 A_WeaponReady(WRF_ALLOWRELOAD) //comment out
		Goto Ready
	Reload:
		TNT1 A 0 A_JumpIfInventory ("VR_RevLoaded",6,"Ready")
		TNT1 A 0 A_JumpIfInventory("Clip",3,"ReloadA")
		Goto OutOfAmmo
	ReloadA:
		TNT1 A 0 A_playsound("wpn/9mmclip1", CHAN_AUTO)
		REVL QRSTUVWXYZ 2
		REVO A 1 A_playsound("wpn/9mmclip2", CHAN_AUTO)
		REVO BCDEFGH 2
		TNT1 A 0 A_playsound("wpn/9mmslide", CHAN_AUTO)
		REVO IJKL 1
		REVL A 1
		goto Bulletin
	Bulletin:
		TNT1 A 0 A_JumpIfInventory("VR_RevLoaded",6,"Ready")
		TNT1 A 0 A_JumpIfInventory("Clip",3,"Bulletin2")
		goto Ready
	Bulletin2:
		TNT1 A 0 A_Giveinventory("VR_RevLoaded",1)
		TNT1 A 0 A_Takeinventory("Clip",3)
		Goto Bulletin
	Flash:
	    TNT1 A 0 A_FireProjectile("BulletTracer",0,0,0,0,FPF_NOAUTOAIM,0)
		REVF BCD 1 Bright A_Light2
		REVF EFGH 1 Bright A_Light1
		Goto LightDone
 	Spawn:
		REVO Q -1
		Stop
OutOfAmmo:
TNT1 A 0 A_PlaySound ("AKEMPT",CHAN_AUTO)
Goto Ready
}
}
ACTOR VR_RevLoaded : Ammo
{  
	Inventory.Amount 0
	Inventory.MaxAmount 6
	Ammo.BackpackAmount 0
	Ammo.BackpackMaxAmount 6
	Inventory.Icon "REVOQ0"
}
