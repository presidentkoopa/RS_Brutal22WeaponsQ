// RSR-BREAK: real VR reload states added (team/reload_break/tools/rsb_patch.py), and the default weapon-switch crash fixed (tool/vplus/vplus_patch.py FIX_SWITCH); otherwise the set's own file.
actor VR_Revolver: Weapon
{
		Weapon.SelectionOrder 1898
		Weapon.AmmoUse 0
		Weapon.AmmoGive1 48
		Weapon.AmmoGive2 6
		Weapon.AmmoType1 "Clip"
		Weapon.AmmoType2 "VR_RevLoaded"
		Tag "Revolver"
	States
	{
	// RSR-BREAK: real VR reload, break-open / revolver (zscript/rsr/rsr_break.zs, team/reload_break/tools/rsb_patch.py).
	// Not in VR, or with Real reload off, RSB_Start goes straight on into the set's own reload.
	RSB_Start:
		REVL A 0 A_TakeInventory("RSB_Go", 1)
		REVL A 0 A_GiveInventory("RSB_Start", 1)
		REVL A 0 A_JumpIfInventory("RSB_Go", 1, "RSB_Open")
		Goto RSB_Stock
	RSB_Open:
		REVL A 0 A_TakeInventory("RSB_Go", 1)
	RSB_Wait:		// the gun as the script shows it ("####" "#"); the script runs every tic
		"####" "#" 1 A_WeaponReady(WRF_NOBOB|WRF_NOFIRE)
		"####" "#" 0 A_GiveInventory("RSB_Tick", 1)
		"####" "#" 0 A_JumpIfInventory("RSB_EjectT", 1, "RSB_Eject")
		"####" "#" 0 A_JumpIfInventory("RSB_LoadT", 1, "RSB_Load")
		"####" "#" 0 A_JumpIfInventory("RSB_Done", 1, "RSB_Done")
		Goto RSB_Wait
	RSB_Eject:		// the ejector threw all six (the set's own frames): the live ones are held until a loader is home
		"####" "#" 0 A_TakeInventory("RSB_EjectT", 1)
	RSB_Eject1:
		"####" "#" 0 A_JumpIfInventory("VR_RevLoaded", 1, 1)
		Goto RSB_Wait
		"####" "#" 0 A_TakeInventory("VR_RevLoaded", 1)
		"####" "#" 0 A_GiveInventory("RSB_Held_VR_Revolver", 1)
		Goto RSB_Eject1
	RSB_Load:		// the speedloader pushed home: the held rounds back, then the set's own Bulletin loop
		"####" "#" 0 A_TakeInventory("RSB_LoadT", 1)
	RSB_Return:
		"####" "#" 0 A_JumpIfInventory("RSB_Held_VR_Revolver", 1, 1)
		Goto RSB_Bulletin
		"####" "#" 0 A_TakeInventory("RSB_Held_VR_Revolver", 1)
		"####" "#" 0 A_GiveInventory("VR_RevLoaded", 1)
		Goto RSB_Return
	RSB_Bulletin:
		"####" "#" 0 A_JumpIfInventory("VR_RevLoaded",6,"RSB_Wait")
		"####" "#" 0 A_JumpIfInventory("Clip",1,"RSB_Bulletin2")
		goto RSB_Wait
	RSB_Bulletin2:
		"####" "#" 0 A_Giveinventory("VR_RevLoaded",1)
		"####" "#" 0 A_Takeinventory("Clip",1)
		Goto RSB_Bulletin
	RSB_Done:		// shut (weapons/9mmslide): the set's own end of a reload
		REVL A 0 A_TakeInventory("RSB_Done", 1)
		Goto Ready
	Ready:
		REVL A 1 A_WeaponReady(WRF_ALLOWRELOAD)
		Loop
	Deselect:
		TNT1 A 0 A_JumpIf(GetCVAR("evw_fastswitch")==0, 2)
		REVL A 1 A_Lower(60)
		Loop
		REVL A 1 A_Lower
		Loop
	Select:
		TNT1 A 0 A_JumpIf(GetCVAR("evw_fastswitch")==0, 2)
		REVL A 1 A_raise(60)
		REVL AAAAAAAAAAAAAAAAAAAAAAAA 1 A_Raise //GUN IS SCREAMING !!
	Fire:
		TNT1 A 0 A_JumpIfInventory ("VR_RevLoaded",1,"CommitExtremeViolence")
		Goto Reload
  CommitExtremeViolence:
		REVL BCD 1
        TNT1 A 0 A_Gunflash
		TNT1 A 0 A_FireBullets (0, 0, -1, 5, "postalpuff")
		TNT1 A 0 A_playsound("weapons/revolver", CHAN_WEAPON)
		TNT1 A 0 A_TakeInventory("VR_RevLoaded",1)
		//gun has been fired
		REVL E 2
		REVL F 2
		REVL GHIJKLMNOP 1 A_WeaponReady(WRF_ALLOWRELOAD) //comment out
		Goto Ready
	Reload:
		TNT1 A 0 A_JumpIfInventory ("VR_RevLoaded",6,"Ready")
		TNT1 A 0 A_JumpIfInventory("Clip",1,"ReloadA")
		Goto OutOfAmmo
	ReloadA:
		REVL A 0 A_Jump(256, "RSB_Start")	// RSR-BREAK: real VR reload (zscript/rsr/rsr_break.zs)
	RSB_Stock:
		TNT1 A 0 A_playsound("weapons/9mmclip1", CHAN_AUTO)
		REVL QRSTUVWXYZ 2
		REVO A 1 A_playsound("weapons/9mmclip2", CHAN_AUTO)
		REVO BCDEFGH 2
		TNT1 A 0 A_playsound("weapons/9mmslide", CHAN_AUTO)
		REVO IJKL 1
		REVL A 1
		goto Bulletin
	Bulletin:
		TNT1 A 0 A_JumpIfInventory("VR_RevLoaded",6,"Ready")
		TNT1 A 0 A_JumpIfInventory("Clip",1,"Bulletin2")
		goto Ready
	Bulletin2:
		TNT1 A 0 A_Giveinventory("VR_RevLoaded",1)
		TNT1 A 0 A_Takeinventory("Clip",1)
		Goto Bulletin
	Flash:
	    TNT1 A 0 A_FireProjectile("BulletTracer",0,0,0,0,FPF_NOAUTOAIM,0)
		REVF BCD 1 Bright A_Light2
		REVF EFGH 1 Bright A_Light1
		Goto LightDone
 	Spawn:
		REVO Q -1
		Stop
OutOfAmmo:
TNT1 A 0 A_PlaySound ("AKEMPT",CHAN_AUTO)
Goto Ready
}
}
ACTOR VR_RevLoaded : Ammo
{  
	Inventory.Amount 0
	Inventory.MaxAmount 6
	Ammo.BackpackAmount 0
	Ammo.BackpackMaxAmount 6
	Inventory.Icon "REVOQ0"
}
