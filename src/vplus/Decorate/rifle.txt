// RS real VR reload states added (team/reload/tools/rsr_patch.py), and the default weapon-switch crash fixed (tool/vplus/vplus_patch.py FIX_SWITCH); otherwise the set's own file.
actor VR_Rifle: Weapon
{
		Weapon.SelectionOrder 1897
		Weapon.AmmoUse 0
		Weapon.AmmoGive1 48
		Weapon.AmmoGive2 20
		Weapon.AmmoType1 "Clip"
		Weapon.AmmoType2 "VR_RifleLoaded"
		Tag "Rifle"
	States
	{
	// RS real VR reload (zscript/rsr/vreload.zs, team/reload/tools/rsr_patch.py). Not in VR,
	// or with Real reload off, RSR_Start goes straight on into the set's own reload.
	RSR_Start:
		RIFL A 0 A_TakeInventory("RSR_Go", 1)
		RIFL A 0 A_GiveInventory("RSR_Start", 1)
		RIFL A 0 A_JumpIfInventory("RSR_Go", 1, "RSR_Out")
		Goto RSR_Stock
	RSR_Out:
		RIFL A 0 A_TakeInventory("RSR_Go", 1)
		RIFL A 0 A_JumpIfInventory("RSR_NoMag_VR_Rifle", 1, "RSR_Wait")	// no magazine in the gun
		RIFL A 0 A_GiveInventory("RSR_MagOut", 1)	// the magazine drops out (weapons/9mmclip1)
	RSR_Unload:		// its rounds are held aside and come back when a magazine is home (the set tops up)
		RIFL A 0 A_JumpIfInventory("VR_RifleLoaded", 1, "RSR_Unload1")
		RIFL A 0 A_GiveInventory("RSR_NoMag_VR_Rifle", 1)
		Goto RSR_Wait
	RSR_Unload1:
		RIFL A 0 A_TakeInventory("VR_RifleLoaded", 1)
		RIFL A 0 A_GiveInventory("RSR_Held_VR_Rifle", 1)
		Goto RSR_Unload
	RSR_Wait:		// the gun as the script shows it ("####" "#"); the script runs every tic
		"####" "#" 1 A_WeaponReady(WRF_NOBOB|WRF_NOFIRE)
		"####" "#" 0 A_GiveInventory("RSR_Tick", 1)
		"####" "#" 0 A_JumpIfInventory("RSR_Seated", 1, "RSR_Fill")
		"####" "#" 0 A_JumpIfInventory("RSR_Done", 1, "RSR_Done")
		Goto RSR_Wait
	RSR_Fill:		// the magazine is home: the set's own Bulletin loop (20 rounds)
		RIFL A 0 A_TakeInventory("RSR_Seated", 1)
		RIFL A 0 A_TakeInventory("RSR_NoMag_VR_Rifle", 1)
	RSR_Return:		// the held rounds first
		RIFL A 0 A_JumpIfInventory("RSR_Held_VR_Rifle", 1, "RSR_Return1")
		Goto RSR_Fill1
	RSR_Return1:
		RIFL A 0 A_TakeInventory("RSR_Held_VR_Rifle", 1)
		RIFL A 0 A_GiveInventory("VR_RifleLoaded", 1)
		Goto RSR_Return
	RSR_Fill1:
		RIFL A 0 A_JumpIfInventory("VR_RifleLoaded", 20, "RSR_Filled")
		RIFL A 0 A_JumpIfInventory("Clip", 1, "RSR_Fill1a")
		Goto RSR_Filled
	RSR_Fill1a:
		RIFL A 0 A_GiveInventory("VR_RifleLoaded", 1)
		RIFL A 0 A_TakeInventory("Clip", 1)
		Goto RSR_Fill1
	RSR_Filled:
		RIFL A 0 A_GiveInventory("RSR_Filled", 1)	// the script: the charging handle
		Goto RSR_Wait
	RSR_Done:
		RIFL A 0 A_TakeInventory("RSR_Done", 1)
		Goto Ready
	Ready:
		RIFL A 1 A_WeaponReady(WRF_ALLOWRELOAD)
		Loop
	Deselect:
		TNT1 A 0 A_JumpIf(GetCVAR("evw_fastswitch")==0, 2)
		RIFL A 1 A_Lower(60)
		Loop
		RIFL A 1 A_Lower
		Loop
	Select:
		TNT1 A 0 A_JumpIf(GetCVAR("evw_fastswitch")==0, 2)
		RIFL A 1 A_raise(60)
		RIFL AAAAAAAAAAAAAAAAAAAAAAAA 1 A_Raise //GUN IS SCREAMING !!
	Fire:
		TNT1 A 0 A_JumpIfInventory ("VR_RifleLoaded",1,"CommitExtremeViolence")
		Goto Reload
  CommitExtremeViolence:
		RIFL B 1
		TNT1 A 0 A_Gunflash
		TNT1 A 0 A_FireBullets (1, 0, -1, 10, "postalpuff2")
		TNT1 A 0 A_playsound("weapons/m16shoot", CHAN_WEAPON)
		//gun has been fired
		TNT1 A 0 A_takeinventory("VR_RifleLoaded", 1)
		RIFL CDE 1

		RIFL F 1 A_WeaponReady(WRF_ALLOWRELOAD) //comment out

		RIFL G 1 A_WeaponReady(WRF_ALLOWRELOAD) //comment out

		RIFL H 1 A_WeaponReady(WRF_ALLOWRELOAD) //comment out

		RIFL I 1 A_WeaponReady(WRF_ALLOWRELOAD) //comment out

		RIFL J 1 A_WeaponReady(WRF_ALLOWRELOAD) //comment out
		//RIFL K 1 A_refire
		RIFL K 1 A_WeaponReady(WRF_ALLOWRELOAD) //comment out
		Goto Ready
	Reload:
		TNT1 A 0 A_jumpifinventory("VR_RifleLoaded", 20, "Ready")
		TNT1 A 0 A_jumpifinventory("Clip", 1, "ReloadA")
		TNT1 A 0 A_PlaySound ("AKEMPT",CHAN_AUTO)
	ReloadA:
		RIFL A 0 A_Jump(256, "RSR_Start")	// RS real VR reload (zscript/rsr/vreload.zs)
	RSR_Stock:
		RIFL LMN 2

		TNT1 A 0 A_takeinventory("RifleFired", 100)
		TNT1 A 0 A_playsound("weapons/9mmclip1", CHAN_AUTO)

		RIFL OPQRSTUVWXYZ 1
		RIFK ABCDEFGHI 2
		RIFK i 0 A_playsound("weapons/9mmclip2", CHAN_AUTO)
		RIFK JKL 1
		RIFK l 0 A_playsound("weapons/9mmslide", CHAN_AUTO)
		RIFK l 0 A_giveinventory("RifleFull", 1)
		RIFK MNO 2
		RIFL A 1
		goto Bulletin
	Bulletin:
		TNT1 A 0 A_JumpIfInventory("VR_RifleLoaded", 20,"Ready")
		TNT1 A 0 A_JumpIfInventory("Clip",1,"Bulletin2")
		goto Ready
	Bulletin2:
		TNT1 A 0 A_Giveinventory("VR_RifleLoaded",1)
		TNT1 A 0 A_Takeinventory("Clip",1)
		Goto Bulletin
		
	Flash:
		TNT1 A 0 A_FireProjectile("BulletTracer",0,0,0,0,FPF_NOAUTOAIM,0)
		RIFF BCD 1 Bright A_Light2
		RIFF EF 1 Bright A_Light1
		Goto LightDone
 	Spawn:
		RIFP A -1
		Stop
OutOfAmmo:
TNT1 A 0 A_PlaySound ("AKEMPT",CHAN_AUTO)
Goto Ready
}
}
ACTOR VR_RifleLoaded : Ammo
{  
	Inventory.Amount 0
	Inventory.MaxAmount 20
	Ammo.BackpackAmount 0
	Ammo.BackpackMaxAmount 20
	Inventory.Icon "RIFFG0"
}
