ACTOR VR_Shotgun : Shotgun
{
  Weapon.AmmoUse 0
  Weapon.AmmoGive1 8
  Weapon.AmmoGive2 0
  Weapon.AmmoType1 "VR_Shell"
  Weapon.AmmoType2 "VR_ShotLoaded"
  
  States
  {
	// RSR-PUMP: hand-loaded shells and a hand-worked pump (zscript/rsr/rsr_pump.zs, tools/rsr_pump_patch.py).
	// Not in VR, with Stock reloading, or with the off hand not on the pump, every hook goes
	// straight on into the set's own states. The shot itself is never touched.
	RSP_Load:		// Reload, after the set's own checks (not full, shells left)
		SHTG A 0 A_TakeInventory("RSP_Go", 1)
		SHTG A 0 A_GiveInventory("RSP_LoadStart", 1)
		SHTG A 0 A_JumpIfInventory("RSP_Go", 1, "RSP_LoadGo")
		Goto StartReload
	RSP_VRIns:	// RSR-PUMP: the VR insert's OWN entry, skipping the set's reload gates.
				// rsr_pump.zs reads the reload button and refuses before sending us here, so by
				// this point there is reserve ammo and room in the tube.
		SHTG A 0 A_TakeInventory("RSP_Insert", 1)
		SHTG A 0 A_TakeInventory("RSP_Go", 1)
		SHTG A 0 A_GiveInventory("RSP_LoadStart", 1)
		SHTG A 0 A_JumpIfInventory("RSP_Go", 1, "RSP_VRIns1")
		Goto Ready					// not in VR, or Stock reloading: the set keeps its own reload
	RSP_VRIns1:
		SHTG A 0 A_TakeInventory("RSP_Go", 1)
		Goto RSP_LoadWait
	RSP_LoadGo:
		SHTG A 0 A_TakeInventory("RSP_Go", 1)
	RSP_LoadWait:	// ReloadLoop: the hand loads shells (rsr_pump.zs); fire ends the reload, as in the set
		SHTG A 1 A_WeaponReady
		SHTG A 0 A_GiveInventory("RSP_LTick", 1)
		SHTG A 0 A_JumpIfInventory("RSP_ShellIn", 1, "RSP_Shell")
		SHTG A 0 A_JumpIfInventory("RSP_Back", 1, "RSP_LBack")
		SHTG A 0 A_JumpIfInventory("RSP_Fwd", 1, "RSP_LFwd")
		Goto RSP_LoadWait
	RSP_Shell:		// a shell pushed home: the set's own InsertShell step
		SHTG A 0 A_TakeInventory("RSP_ShellIn", 1)
		SHTG A 0 A_GiveInventory("VR_ShotLoaded", 1)
		SHTG A 0 A_TakeInventory("VR_Shell", 1)
		SHTG A 0 A_JumpIfInventory("VR_ShotLoaded", 8, "ReloadEnd")	// ReloadLoop's own checks
		SHTG A 0 A_JumpIfInventory("VR_Shell", 1, "RSP_LoadWait")
		Goto OutOfAmmo
	RSP_LBack:		// the hand has the pump back while loading (the set never pumps in its reload)
		SHTG A 0 A_TakeInventory("RSP_Back", 1)
		Goto RSP_LoadWait
	RSP_LFwd:		// forward again: the reload ends (ReloadEnd)
		SHTG A 0 A_TakeInventory("RSP_Fwd", 1)
		Goto ReloadEnd
	RSP_Pump:		// after the shot's frames (BCDEFG HI), before the set's pump (J-Z)
		SHTG A 0 A_TakeInventory("RSP_Go", 1)
		SHTG A 0 A_GiveInventory("RSP_PumpStart", 17)	// 17: the set's own J-Z, to its refire
		SHTG A 0 A_JumpIfInventory("RSP_Go", 1, "RSP_PGo")
		Goto RSP_PumpStock
	RSP_PGo:
		SHTG A 0 A_TakeInventory("RSP_Go", 1)
	RSP_PWait:		// the hand works the pump (rsr_pump.zs moves it, plays shotpump and ejects the shell)
		SHTG A 1 A_WeaponReady(WRF_NOFIRE|WRF_NOBOB|WRF_NOSECONDARY|WRF_ALLOWRELOAD)
		SHTG A 0 A_GiveInventory("RSP_PTick", 1)
		SHTG A 0 A_JumpIfInventory("RSP_Back", 1, "RSP_PBack")
		SHTG A 0 A_JumpIfInventory("RSP_Fwd", 1, "RSP_PFwd")
		SHTG A 0 A_JumpIfInventory("RSP_Insert", 1, "RSP_VRIns")	// a shell, mid pump wait
		SHTG A 0 A_JumpIfInventory("RSP_Auto", 1, "RSP_PAuto")
		SHTG A 0 A_JumpIfInventory("RSP_Ready", 1, "RSP_PReady")
		Goto RSP_PWait
	RSP_PBack:
		SHTG A 0 A_TakeInventory("RSP_Back", 1)
		Goto RSP_PWait
	RSP_PFwd:
		SHTG A 0 A_TakeInventory("RSP_Fwd", 1)
		SHTG A 0 A_JumpIfInventory("RSP_Ready", 1, "RSP_PReady")
		Goto RSP_PWait		// ready once the set's own refire time is up (RSP_Ready)
	RSP_PReady:
		SHTG A 0 A_TakeInventory("RSP_Ready", 1)
		Goto Ready
	RSP_PAuto:		// the hand left the pump before working it: the set's own pump
		SHTG A 0 A_TakeInventory("RSP_Auto", 1)
		Goto RSP_PumpStock
  Ready:
    SHTG A 0 A_JumpIfInventory("RSP_Insert", 1, "RSP_VRIns")	// RSR-PUMP: the hand insert
    SHTG A 1 A_WeaponReady(WRF_ALLOWRELOAD)
    Loop

  Deselect:
	TNT1 A 0 A_JumpIf(GetCVAR("evw_fastswitch")==0, 2)
    SHTG A 1 A_Lower(60)
    Loop
    SHTG A 1 A_Lower
    Loop

  Select:
	TNT1 A 0 A_JumpIf(GetCVAR("evw_fastswitch")==0, 2)
    SHTG A 1 A_Raise(60)
    Loop
    SHTG A 1 A_Raise
    Loop

  Fire:
    TNT1 A 0 A_JumpIfInventory("VR_ShotLoaded", 1, "CommitExtremeViolence")
    Goto Reload

  CommitExtremeViolence:
    TNT1 A 0 A_JumpIfInventory("VR_ShotLoaded", 1, "ACommitExtremeViolence")
    TNT1 A 0 A_PlaySound("AKEMPT", CHAN_AUTO)
    Goto Ready

  ACommitExtremeViolence:
    TNT1 A 0 A_TakeInventory("VR_ShotLoaded", 1)
    SHTG A 0 A_FireBullets(5.6, 0, 7, 5, "vrshotpuff")
    SHTG A 0 A_PlaySound("wpn/shotgun1", CHAN_WEAPON)
    SHTG A 0 A_GunFlash
    SHTG BCDEFG 2
    SHTG HI 1
		Goto RSP_Pump	// RSR-PUMP: a hand-worked pump (zscript/rsr/rsr_pump.zs)
	RSP_PumpStock:
	SHTG A 0 A_PlaySound("shotpump", CHAN_WEAPON)
	SHTG JKLMNOPQRSTUVWXYZ 1
    Goto Ready

  Flash:
    TNT1 AAAAAA 0 A_FireProjectile("BulletTracer",random(-3,3),0,0,0,FPF_NOAUTOAIM,random(-1,1))
    SHTF ABC 1 Bright A_Light2
    SHTF DEF 1 Bright A_Light1
    Goto LightDone

  Spawn:
    SHOT A -1
    Stop

  // ========================
  // RELOAD (NON-INTERRUPT)
  // ========================
  Reload:
    // If full, don't reload
    TNT1 A 0 A_JumpIfInventory("VR_ShotLoaded", 8, "Ready")
    // If no spare shells, don't reload
    TNT1 A 0 A_JumpIfInventory("VR_Shell", 1, "RSP_Load")	// RSR-PUMP: hand-loaded shells (was: StartReload)
    Goto OutOfAmmo

  StartReload:
    // Start animation
    SHTG A 3
    Goto ReloadLoop

  ReloadLoop:
    // INTERRUPT RELOAD IF PLAYER FIRES
    SHTG A 1 A_WeaponReady(WRF_ALLOWRELOAD)
    // Check if full
    TNT1 A 0 A_JumpIfInventory("VR_ShotLoaded", 8, "ReloadEnd")
    // Check if out of ammo
    TNT1 A 0 A_JumpIfInventory("VR_Shell", 1, "InsertShell")
    Goto OutOfAmmo

  InsertShell:
    // Insert shell animation + sound
    SHTG RQPONMLKJ 1
    TNT1 A 0 A_PlaySound("Shotgun/Reload", CHAN_AUTO)
    TNT1 A 0 A_GiveInventory("VR_ShotLoaded", 1)
    TNT1 A 0 A_TakeInventory("VR_Shell", 1)
    // Loop back until full or empty
    Goto ReloadLoop

  ReloadEnd:
    // Finish reload animation
    SHTG A 3
    Goto Ready

  OutOfAmmo:
    TNT1 A 0 A_PlaySound("AKEMPT", CHAN_AUTO)
    Goto Ready
  }
}

  Inventory.Amount 0
  Inventory.MaxAmount 8
  Ammo.BackpackAmount 0
  Ammo.BackpackMaxAmount 8
  Inventory.Icon "SHOTA0"
}

ACTOR VR_SuperShotgun : SuperShotgun
{
  Weapon.AmmoType "VR_Shell"
  States
  {
	// RSR-BREAK: real VR reload, break-open / revolver (zscript/rsr/rsr_break.zs, team/reload_break/tools/rsb_patch.py).
	// Not in VR, or with Real reload off, RSB_Start goes straight on into the set's own reload.
	RSB_Start:
		SHT2 A 0 A_TakeInventory("RSB_Go", 1)
		SHT2 A 0 A_GiveInventory("RSB_Start", 1)
		SHT2 A 0 A_JumpIfInventory("RSB_Go", 1, "RSB_Open")
		Goto RSB_Stock
	RSB_Open:
		SHT2 A 0 A_TakeInventory("RSB_Go", 1)
	RSB_Wait:		// the gun as the script shows it ("####" "#"); the script runs every tic
		"####" "#" 1 A_WeaponReady(WRF_NOBOB|WRF_NOFIRE)
		"####" "#" 0 A_GiveInventory("RSB_Tick", 1)
		"####" "#" 0 A_JumpIfInventory("RSB_EjectT", 1, "RSB_Eject")
		"####" "#" 0 A_JumpIfInventory("RSB_LoadT", 1, "RSB_Load")
		"####" "#" 0 A_JumpIfInventory("RSB_Done", 1, "RSB_Done")
		Goto RSB_Wait
	RSB_Eject:		// (its spent shells are thrown out by the script, as its own frames throw them)
		"####" "#" 0 A_TakeInventory("RSB_EjectT", 8)
		Goto RSB_Wait
	RSB_Load:		// one shell pushed home in a barrel: the set's own count (VR_SSGAlt, the loaded barrels; its reload gives 2)
		"####" "#" 0 A_TakeInventory("RSB_LoadT", 1)
		"####" "#" 0 A_GiveInventory("VR_SSGAlt", 1)
		Goto RSB_Wait
	RSB_Done:		// shut (weapons/sshotc): the set's own end of a reload
		SHT2 A 0 A_TakeInventory("RSB_Done", 1)
		Goto Ready
  Ready:
    SHT2 A 1 A_WeaponReady(WRF_ALLOWRELOAD)
    Loop
  Deselect:
	TNT1 A 0 A_JumpIf(GetCVAR("evw_fastswitch")==0, 2)
    SHT2 AA 1 A_Lower(60)
    Loop
  Select:
TNT1 A 0 A_JumpIf(GetCVAR("evw_fastswitch")==0, 2)
    SHT2 AA 1 A_Raise(60)
    Loop
  Fire:
	TNT1 A 0 A_JumpIfInventory ("VR_SSGLoaded",1,"CommitExtremeViolence")
	TNT1 A 0 A_JumpIfInventory ("VR_SSGAlt",1,"ACommitExtremeViolence")
	Goto Reload
  CommitExtremeViolence:
	TNT1 A 0 A_TakeInventory("VR_SSGLoaded", 1)
	TNT1 A 0 A_TakeInventory("VR_SSGAlt", 2)
    SHT2 A 0 A_FireBullets (11.2, 7.1, 20, 5, "vrssgpuff")
    SHT2 A 0 A_PlaySound ("wpn/1shotgun", CHAN_AUTO)
    SHT2 A 2 A_GunFlash
    SHT2 BCDEFGH 2
	Goto Reload
	
  AltFire:
	TNT1 A 0 A_JumpIfInventory ("VR_SSGAlt",1,"ACommitExtremeViolence")
	Goto Reload
  ACommitExtremeViolence:
	TNT1 A 0 A_TakeInventory("VR_SSGLoaded", 1)
	TNT1 A 0 A_TakeInventory("VR_SSGAlt", 1)
	TNT1 A 0 A_TakeInventory("VR_Shell", 1)
    SHTG A 0 A_FireBullets (5.6, 0, 7, 5, "vrshotpuff")
    SHTG A 0 A_PlaySound ("wpn/shotgun1", CHAN_WEAPON)
    SHT2 A 2 A_GunFlash
    SHT2 BC 2
	SHT2 DEFGH 2 A_WeaponReady(WRF_ALLOWRELOAD)
	SHT2 I 10 A_WeaponReady(WRF_ALLOWRELOAD) // cooldown frame
	
	// Auto-reload trigger if both chambers are empty
    TNT1 A 0 A_JumpIfInventory("VR_SSGLoaded", 1, "Ready") // if still loaded, stay ready
    TNT1 A 0 A_JumpIfInventory("VR_SSGAlt", 1, "Ready")    // if alt chamber still loaded, stay ready
    Goto Reload
	
  Reload:
	TNT1 A 0 A_JumpIfInventory ("VR_SSGLoaded",1,"Ready")
    SHT2 I 0 A_CheckReload
    SHT2 I 0 A_Jump(256, "RSB_Start")	// RSR-BREAK: real VR reload (zscript/rsr/rsr_break.zs)
  RSB_Stock:
    SHT2 I 2 A_OpenShotgun2
    SHT2 JKLMNOPQR 3 
	TNT1 A 0 A_GiveInventory("VR_SSGLoaded", 1)
	TNT1 A 0 A_GiveInventory("VR_SSGAlt", 2)
    SHT2 S 2 A_LoadShotgun2
    SHT2 TUV 3
    SHT2 X 2 A_CloseShotgun2 
    SHT2 Y 2
    SHT2 Y 1 A_ReFire
    Goto Ready
  Flash:
    TNT1 AAAAAA 0 A_FireProjectile("BulletTracer",random(-3,3),0,0,0,FPF_NOAUTOAIM,random(-1,1))
    SHT4 AB 1 Bright A_Light2
    SHT4 CD 1 Bright A_Light1
    Goto LightDone
  Spawn:
    SGN2 A -1
    Stop
  }
}
