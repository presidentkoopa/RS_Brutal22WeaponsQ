// ============================================================================
// RS_QuestHands -- hand models
//
// MODELDEF keys a model on (caller class, sprite, frame) -- models.cpp:1165:
//   if (smff->type == ti && smff->sprite == sprite && smff->frame == frame)
// It matches the EXACT class, never a child, and an unknown class name is fatal
// at startup (models.cpp:750-754). So these blocks name QH_Hands and QH_GunSide
// outright, and each sprite frame used has a real lump in sprites/.
//
// One block per sprite frame rather than one block with several models, because
// Scale is per block and because disjoint frames in a shared block is the
// fiddlier reading of the parser.
//
// SCALE IS DERIVED, NOT GUESSED. The hook's frame is 1 model unit = 1 cm times
// this scale (models.cpp:225-277 with hw_openxrdevice.cpp:442: 0.01 *
// vr_vunits_per_meter 34 = 0.34 map units per model unit, and 34 units = 1 m).
//
//   python tools/md3_pose_survey.py models/hand.md3 --bbox
//
// hand.md3's longest extent is 25.16 model units, so at Scale 1.0 the hand
// would be 25 cm across -- about a third too big. 0.75 puts it near 18.9 cm,
// which is an adult hand. All three meshes share the scale because they are the
// same pair of hands and have to match each other.
//
// This is a derived starting number, not a measurement of what looks right.
// Seating is still the owner's to feel; every offset is a slider.
//
// Assets: Rusted Legacy hands, MIT per models/LICENSE_RustedLegacy_MIT.txt.
// Its README calls the art "for education purpose only" -- the licence question
// in the plan's "Not known yet" is still open and is not settled by this file.
// ============================================================================


// ---- free off hand : owned by QH_Hands -> OFF controller -------------------
// hand.md3 is the mesh meant for a hand drawn at its own controller. The engine
// mirrors whatever lands on the physical left controller by itself
// (hw_vrmodes.cpp:468-469), so this one mesh serves both free hands -- unless
// +WEAPON.NO_AUTO_REVERSE is set on the off-hand weapon, which suppresses it.
Model QH_Hands
{
	Path "models"
	Model 0 "hand.md3"
	Skin 0 "hand.png"
	Scale 0.75 0.75 0.75
	// Frame 0 -- open relaxed. Owner's pick, 2026-10-03.
	FrameIndex QHND B 0 0
	// Frame 209 -- closed round the magazine: a wrap with a gap through it, not
	// a fist. Owner confirmed 2026-10-03; frame 5 is the true fist if one is
	// ever wanted. This is step 4's blend target, so B and D share a block --
	// they must, since the engine blends between frames of the SAME model.
	FrameIndex QHND D 0 209
}

// shell_12ga.md3 from RS_Ballistics, mined with the owner's go-ahead
// 2026-10-03. Four surfaces with their own skins, so SurfaceSkin rather than
// Skin.
//
// UNITS DIFFER BETWEEN THE TWO LIBRARIES AND THIS MATTERS:
//   RS_VR_Weapons meshes are in CENTIMETRES  -> Scale 1.0 here
//   RS_Ballistics meshes are in MILLIMETRES  -> Scale 0.1 here
// The hook's space is 1 model unit = 1 cm times this scale, so a 70.06-unit
// mesh needs 0.1 to read as a 7.0 cm shell. RS_Ballistics' own MODELDEF uses
// 0.03279 for these, but that is world space for a dropped casing, not ours.
//
// The RS_Ballistics casings are modelled to real cartridge dimensions:
// 9mm 19.20, .45 22.84, .357 33.06, 5.56 44.70, 12ga 70.06 -- all correct to
// the millimetre, which is why they are worth taking over a generic cylinder.
Model QH_Hands
{
	Path "models"
	Model 0 "shell_12ga.md3"
	SurfaceSkin 0 0 "casing_brass.png"
	SurfaceSkin 0 1 "shell_hull.png"
	SurfaceSkin 0 2 "shell_inner.png"
	SurfaceSkin 0 3 "casing_primer.png"
	Scale 0.1 0.1 0.1
	FrameIndex QHND F 0 0
}

// ---- gun hand : owned by QH_GunSide -> MAIN controller ---------------------
// hand2.md3 is the mesh meant for the hand drawn on the gun. 196 frames.
//
// IT IS NOT A POSE LIBRARY. Measured with tools/md3_part_track.py, its centroid
// travels X 3.9..49.1 and Y -14.9..41.5 across its frames -- it is an animated
// motion sequence authored in gun space. hand.md3 by contrast stays inside
// X -9.4..4.7 for all 211 of its frames, authored at the origin.
//
// So for THIS mesh, choosing a frame chooses a POSITION as much as a pose, and
// the hook's offset can only correct what is left over. Pick the grip frame
// first, then trim with qh_gun_*. Frame 0 is a placeholder, not a choice.
Model QH_GunSide
{
	Path "models"
	Model 0 "hand2.md3"
	Skin 0 "hand.png"
	Scale 0.75 0.75 0.75
	// Frame 29 -- three fingers wrapped, index along the frame. Chosen from
	// 28/29/30/31 as the mid-range one, 2026-10-03.
	//
	// The frame carries position as well as pose on this mesh, so qh_gun_*
	// defaults in CVARINFO.txt cancel THIS frame's centroid. Changing the frame
	// means changing those three. Measured values:
	//     28  ->  -11.327   5.759  -1.160
	//     29  ->  -12.770   3.767  -1.156
	//     30  ->  -14.865   1.341  -1.087
	//     31  ->  -16.282  -0.119  -1.032
	FrameIndex QHGS A 0 29
}
