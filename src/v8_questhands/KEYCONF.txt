// THE BASELINE. Run this before measuring anything.
//
// vr_weaponScale and vr_3dweaponOffsetX/Y/Z sit INSIDE the weapon transform
// (models.cpp:251 and :277), so whatever the player has set silently translates
// and scales every hand placement. Numbers measured without zeroing them are
// right on one machine and wrong everywhere else.
//
// ZScript cannot write these: they are plain user cvars with no CVAR_MOD, and
// vmnatives.cpp:984-990 throws on a write outside menu code. A console alias
// sidesteps that entirely, which is why this is here and not in script.
alias qh_baseline "set vr_3dweaponOffsetX 0; set vr_3dweaponOffsetY 0; set vr_3dweaponOffsetZ 0; set vr_weaponScale 1.0; echo RS_QuestHands: weapon offsets zeroed, scale 1.0"

// What the player had before, so the baseline is not a one-way door.
alias qh_baseline_show "echo vr_weaponScale $vr_weaponScale / offX $vr_3dweaponOffsetX offY $vr_3dweaponOffsetY offZ $vr_3dweaponOffsetZ"

// Step 1's go/no-go readout: which layers exist, who owns them, which hand each
// is on, and the metric-corrected hand separation.
alias qh_report "netevent qh_report"
