// ============================================================================
// SCOPES, AND WHY EACH ONE IS WHAT IT IS
//
// Quest has no multiplayer today, but the pack is written to survive its
// forthcoming update, and cvar scope is where a mod usually fails that quietly.
//
//   server  -- read in PLAY scope, so it decides something the simulation does.
//              Every machine must agree or the games diverge. A reload that
//              releases at 0.35 m on one player's machine and 0.20 m on
//              another's is a desync, not a preference.
//   user    -- read only in the `ui` hook. It moves pixels on one screen and
//              touches no simulation state, so it is per-player taste and MUST
//              stay user: forcing everyone's hand offsets to match would be a
//              worse bug than the one it fixed.
//
// The split is not a judgement call -- it follows the ui/play scope the engine
// already enforces. If a value crosses from ui to play, its scope has to move
// with it.
// ============================================================================

// ============================================================================
// RS_QuestHands -- placement cvars
//
// Every number the hook returns is a slider, because none of them can be
// derived offline: where a hand sits on a controller is the owner's to feel in
// the headset and report. Units are MODEL units in the hook's frame --
// X right, Y up, Z back toward the player, 1 unit = 1 cm times the MODELDEF
// scale (models.cpp:225-277, hw_openxrdevice.cpp:442).
//
// That frame is isotropic. Do NOT apply the pixelstretch correction to these;
// it belongs only to map-space hand positions. See zscript/qh_metric.zs.
//
// All `user` so the hook can read them from ui scope. Note that CVARINFO
// `nosave` would still write to the ini -- it only skips savegames and the
// network -- so there is no point reaching for it here.
// ============================================================================

// ---- free off hand : rides the off controller ----
user float qh_free_x = 0.0;
user float qh_free_y = 0.0;
user float qh_free_z = 0.0;
user float qh_free_yaw = 0.0;
user float qh_free_pitch = 0.0;
user float qh_free_roll = 0.0;

// ---- carried magazine / shell : sits in the free hand, offset from it ----
user float qh_carry_x = 0.0;
user float qh_carry_y = 0.0;
user float qh_carry_z = 0.0;

// ---- gun hand : rides the main controller, offset to the gun's grip ----
//
// These defaults cancel hand2.md3 frame 29's own centroid, measured with
// tools/md3_part_track.py and multiplied by the MODELDEF Scale of 0.75. That
// mesh's frames carry position as well as pose, so the three are COUPLED TO THE
// FRAME -- change QHGS A in MODELDEF.txt and these must change with it. The
// other candidates' values are listed there.
//
// With these, the hand starts at the controller instead of flung 12 units off
// it, and the sliders are for seating rather than for hunting.
user float qh_gun_x = -12.770;
user float qh_gun_y = 3.767;
user float qh_gun_z = -1.156;
user float qh_gun_yaw = 0.0;
user float qh_gun_pitch = 0.0;
user float qh_gun_roll = 0.0;

// ---- support hand : rides the main controller, at the gun's support point ----
//
// Cancels hand2.md3 frame 13's centroid, same method, same coupling.
user float qh_sup_x = -8.334;
user float qh_sup_y = -7.182;
user float qh_sup_z = -2.388;
user float qh_sup_yaw = 0.0;
user float qh_sup_pitch = 0.0;
user float qh_sup_roll = 0.0;

// ---- steadying stale-latch threshold, in REAL METRES ----
//
// gl_openvr.cpp:1717-1728 latches steadying on the off-hand grip press and only
// checks the 0.50 m gap at that instant; gl_openvr.cpp:1704 then gates the whole
// block on menuactive == MENU_Off, so releasing the grip inside a menu leaves
// the flag stuck on with no button held. Hands further apart than this, while
// the flag claims we are steadying, are treated as a stale latch.
//
// Above the engine's own 0.50 m engage distance on purpose.
server float qh_stale_latch_m = 0.60;

// ---- step 2 reload gestures, in REAL METRES ----
//
// Hand separation, metric-corrected (qh_metric.zs). Gestures are by position,
// never the off-hand grip button -- the engine takes that button for steadying
// (gl_openvr.cpp:1717), and steadying and reloading are mutually exclusive.
//
// VR_Rifle's three stops: magazine clear of the well (RIFL Z), about to insert
// (RIFK A), about to rack (RIFK J).

// Magazine out releases once the off hand has gone this far out to fetch.
server float qh_fetch_m = 0.35;

// Magazine in releases once the off hand is back this close to the well.
// Must be comfortably below qh_fetch_m or the two will chatter.
server float qh_insert_m = 0.18;

// The rack releases once the off hand pulls back this far again.
server float qh_rack_m = 0.30;

// ---- step 3: where the magazine sits in the support hand ----
//
// The ride track itself is MEASURED from the gun's mesh
// (tools/md3_part_track.py, surface weapon.005 of m16.md3) and is not a slider.
// These three are, because where in the HAND the magazine sits is a property of
// the hand mesh, not of the gun.
user float qh_ride_x = 0.0;
user float qh_ride_y = 0.0;
user float qh_ride_z = 0.0;

// ---- shells per insert ----
//
// Owner, 2026-10-03: "no matter what set we are playing i want to be able to do
// 1 to four shells per insert to not wear myself out."
//
// A tube-fed gun would otherwise want eight fetch-and-return trips to fill. One
// trip out earns this many credits; each shell spends one; the hand only travels
// again when they run out. 1 is strictly one shell per trip.
//
// Applies to every QHR_HAND_CYCLE stop in every set, not just the shotgun.
server float qh_shells_per_insert = 1;

// ---- enhanced ballistics ----
//
// Owner, 2026-10-03: "my ideal is 'enhanced ballistics: on / off' and off would
// be BD standard ballistics, projectiles, etc."
//
// OFF IS A REAL OFF. QH_Ballistics returns on its first line and nothing of the
// loaded mod is touched -- no replaces, no owned actors, no altered shots. On,
// it only adds to impacts that have already happened.
//
// Default on, in keeping with effects defaulting on and high.
server int qh_enhanced = 1;

// How much debris an impact throws, as a multiplier on the per-surface count.
// 0 silences debris while leaving the rest of enhanced mode on.
server float qh_debris_scale = 1.0;

// ---- projectile trails ----
//
// Trails are ENGINE PARTICLES, not actors: a particle has no thinker, no
// blockmap entry and no state machine. An actor per trail segment would be
// hundreds of thinkers a second off a plasma rifle and would not survive Quest.
//
// Punch comes from contrast and brevity rather than volume -- a short bright
// core that fades in a few tics reads harder than a long dim smear and costs a
// fraction as much.
//
// All server, because they gate playsim spawns. See the scope note at the top.

// Most projectiles that may carry a trail at once. Past this, new ones get none.
server int qh_trail_max = 24;

// Emit a particle every N tics. 1 = every tic. Raise it to thin trails out.
server int qh_trail_every = 1;

// Particle lifetime in tics. 0 silences trails and leaves impacts alone.
server int qh_trail_life = 7;

// Particle size.
server float qh_trail_size = 2.0;

// ============================================================================
// REAL GRABS -- see zscript/qh_grab.zs for why these are tunable and not
// measured. The drawn gun goes through a transform script cannot read, so a
// grab point is placed by eye against the marker and then baked into the card.
//
// RUN `qh_baseline` BEFORE TUNING. vr_weaponScale and vr_3dweaponOffsetX/Y/Z
// sit inside the weapon transform, so numbers found without zeroing them are
// right on one machine and wrong everywhere else.
// ============================================================================

// Master switch. 0 falls back to the old hand-separation gesture everywhere,
// which is the escape hatch if a grab point turns out to be badly placed.
server int qh_grab = 1;

// How far the off hand must get from the gun to count as having LEFT it, for
// the "hand goes out to fetch" stop. Measured from the main hand, so any
// direction counts.
server float qh_grab_body_m = 0.30;

// ----------------------------------------------------------------------------
// THE GLOBAL DEFAULT PART POSITIONS, in metres along the gun's own axes:
// x forward along the bore, y to the gun's right, z up.
//
// One set shared by every gun that has no baked geometry of its own. That is
// deliberately crude -- a pistol's magazine well is not where a rifle's is --
// but it means every carded gun gets a real grab immediately and can be tuned,
// instead of 13 guns waiting on 13 bakes. Bake per gun as they get tuned.
//
// THESE STARTING VALUES ARE A GUESS, stated plainly: roughly a hand's width
// forward of the grip and a little below the bore, which is where a rifle
// magazine lives. Nothing has measured them.
// ----------------------------------------------------------------------------
server float qh_well_x = 0.10;
server float qh_well_y = 0.00;
server float qh_well_z = -0.08;
server float qh_well_r = 0.12;

// The charging handle / bolt / pump. Slightly forward, to the gun's right.
server float qh_handle_x = 0.02;
server float qh_handle_y = 0.04;
server float qh_handle_z = 0.02;
server float qh_handle_r = 0.10;

// Show the grab spheres as particles while tuning. An instrument, not a
// feature, so it is off by default. Green ring = your off hand is inside it.
server int qh_grab_marker = 0;
